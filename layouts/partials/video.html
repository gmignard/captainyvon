{{- /* Facade video : aucune iframe au chargement, donc aucune requete vers un domaine tiers et aucun cookie avant action de l'utilisateur. Le bouton de lecture EST un lien direct vers la video : sans JavaScript il ouvre la plateforme, avec JavaScript il injecte l'iframe sur place. */ -}} {{- $page := . -}} {{- with .Params.video -}} {{- $id := .id -}} {{- if $id -}} {{- $plateforme := lower (.plateforme | default "youtube") -}} {{- $ratio := .ratio | default "16:9" -}} {{- $ratioCSS := replace (replace $ratio ":" " / ") "x" " / " -}} {{- $embed := "" -}} {{- $direct := "" -}} {{- $nomPlateforme := "" -}} {{- if eq $plateforme "vimeo" -}} {{- $embed = printf "https://player.vimeo.com/video/%s?dnt=1&autoplay=1" $id -}} {{- $direct = printf "https://vimeo.com/%s" $id -}} {{- $nomPlateforme = "Vimeo" -}} {{- else -}} {{- $embed = printf "https://www.youtube-nocookie.com/embed/%s?autoplay=1&rel=0" $id -}} {{- $direct = printf "https://www.youtube.com/watch?v=%s" $id -}} {{- $nomPlateforme = "YouTube" -}} {{- end -}} {{- /* Poster : la couverture du projet, jamais une vignette distante. */ -}} {{- $couverture := "" -}} {{- with $page.Params.couverture }}{{ $couverture = $page.Resources.GetMatch . }}{{ end -}} {{- if not $couverture }}{{ $couverture = $page.Resources.GetMatch "couverture.*" }}{{ end -}} {{- $poster := "" -}} {{- with $couverture -}} {{- $l := cond (gt .Width 1600) 1600 .Width -}} {{- $poster = .Resize (printf "%dx webp q82" $l) -}} {{- end }}
{{- with $poster }} {{- end }}

Ouvrir la vidéo sur {{ $nomPlateforme }}

{{- end -}} {{- end -}}
