{{- /* Galerie du projet. La selection et l'ordre viennent du champ `galerie` de l'entete : une liste de noms de fichiers du page bundle. Sans ce champ, aucune galerie n'est rendue — le contenu du dossier ne pilote plus l'affichage. Images pleine largeur du contenu, empilees. Traitement Hugo Image Processing : WebP, largeurs 800 et 1600, sans jamais agrandir la source. Chargement differe sauf la premiere image. */ -}} {{- $page := . -}} {{- $images := slice -}} {{- range .Params.galerie -}} {{- with $page.Resources.GetMatch . }}{{ $images = $images | append . }}{{ end -}} {{- end -}} {{- $titre := $page.Title -}} {{- if not $titre }}{{ with $page.File }}{{ $titre = humanize .ContentBaseName }}{{ end }}{{ end -}} {{- with $images }}
{{- range $i, $r := . -}} {{- $l800 := cond (gt $r.Width 800) 800 $r.Width -}} {{- $l1600 := cond (gt $r.Width 1600) 1600 $r.Width -}} {{- $petite := $r.Resize (printf "%dx webp q82" $l800) -}} {{- $grande := cond (gt $l1600 $l800) ($r.Resize (printf "%dx webp q82" $l1600)) $petite -}} {{- /* Les images occupent toute la largeur du contenu, elle-meme plafonnee a --largeur-max moins la marge laterale. */ -}} {{- $sizes := "(min-width: 1740px) 1600px, 92vw" -}} {{- /* Alt derive du titre du projet, numerote pour rester distinct. */ -}} {{- $alt := printf "%s — image %d" $titre (add $i 1) -}} {{- end }}
{{- end -}}
